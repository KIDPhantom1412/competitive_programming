# 数据结构模板

## 树状数组

### 单点修改，区间查询

```cpp
const int N = int(1e5) + 5;

using LL = long long;

int n;
LL tr[N];

void update(int x, LL d) { // 单点修改：A[x] += d
    for (int i = x; i <= n; i += i & -i) {
        tr[i] += d;
    }
}

LL sum(int x) { // 前缀和查询：A[1..x]
    LL res = 0;
    for (int i = x; i; i -= i & -i) {
        res += tr[i];
    }
    return res;
}

LL sum(int l, int r) { // 区间查询：A[l..r]
    return sum(r) - sum(l - 1);
}
```

- 用 `LL`（`long long`）存储和返回值，防止累加结果溢出（如逆序对计数最大约 $n^2/2$）。
- 多测时记得清空：`std::fill(tr, tr + n + 5, 0);`

### 区间修改，单点查询

设 `D` 为 `A` 的差分数组（`D[i] = A[i] - A[i - 1]`）。区间加 `[l, r]` 等价于 `D[l] += d`、`D[r + 1] -= d`，单点值即差分前缀和：`A[x] = D[1] + ... + D[x]`。

```cpp
const int N = int(1e5) + 5;

using LL = long long;

int n;
LL tr[N];

void update(int x, LL d) {
    for (int i = x; i <= n; i += i & -i) {
        tr[i] += d;
    }
}

LL sum(int x) {
    LL res = 0;
    for (int i = x; i; i -= i & -i) {
        res += tr[i];
    }
    return res;
}

void update(int l, int r, LL d) { // 区间修改：A[l..r] += d
    update(l, d), update(r + 1, -d);
}

// 单点查询：sum(x) 即 A[x] 的最终值
```

- 多测时记得清空：`std::fill(tr, tr + n + 5, 0);`

### 区间修改，区间查询

设 `D` 为 `A` 的差分数组，`PS(x) = A[1] + ... + A[x]` 为前缀和。展开可得 `PS(x) = Σ D[j]·(x - j + 1)`，用两个树状数组分别维护：

- `t1` 存 `D[j]` 的前缀和，记 `S1(x) = Σ D[j]`（`j ≤ x`）
- `t2` 存 `(j - 1) * D[j]` 的前缀和，记 `S2(x) = Σ (j - 1)·D[j]`（`j ≤ x`）

则前缀和公式：`PS(x) = x * S1(x) - S2(x)`，区间和 `[l, r] = PS(r) - PS(l - 1)`。

```cpp
const int N = int(1e5) + 5;

using LL = long long;

int n;
LL t1[N], t2[N];

void update(LL* tr, int x, LL d) {
    for (int i = x; i <= n; i += i & -i) {
        tr[i] += d;
    }
}

LL sum(LL* tr, int x) {
    LL res = 0;
    for (int i = x; i; i -= i & -i) {
        res += tr[i];
    }
    return res;
}

LL sum(int x) { // 前缀和查询：A[1..x]
    return x * sum(t1, x) - sum(t2, x);
}

void update(int l, int r, LL d) { // 区间修改：A[l..r] += d
    update(t1, l, d), update(t1, r + 1, -d);
    update(t2, l, (l - 1) * d), update(t2, r + 1, -r * d);
}

LL sum(int l, int r) { // 区间查询：A[l..r]
    return sum(r) - sum(l - 1);
}
```

- 建树时直接 `update(i, i, A[i])`，不要再手动差分。

### 二维：矩形修改，矩形查询

设 `D` 为矩阵 `A` 的二维差分数组（`A[i][j] = Σ_{p≤i, q≤j} D[p][q]`）。矩形加 `(a, b)-(c, d)` 等价于对 `D` 做 4 个单点更新：`D[a][b] += d`、`D[a][d+1] -= d`、`D[c+1][b] -= d`、`D[c+1][d+1] += d`。

二维前缀和展开：

$$
S(x, y) = \sum_{i \le x,\ j \le y} (x - i + 1)(y - j + 1) \cdot D_{i,j}
        = (x+1)(y+1) \sum D - (y+1) \sum i \cdot D - (x+1) \sum j \cdot D + \sum i \cdot j \cdot D
$$

用 4 个二维树状数组分别维护 `D`、`i·D`、`j·D`、`i·j·D` 的前缀和，矩形求和用前缀和容斥。

```cpp
const int N = 2060;

using LL = long long;

int n, m;
// t1 = D, t2 = i*D, t3 = j*D, t4 = i*j*D
LL t1[N][N], t2[N][N], t3[N][N], t4[N][N];

void update(LL tr[N][N], int x, int y, LL d) {
    for (int i = x; i <= n; i += i & -i) {
        for (int j = y; j <= m; j += j & -j) {
            tr[i][j] += d;
        }
    }
}

LL sum(LL tr[N][N], int x, int y) {
    LL res = 0;
    for (int i = x; i; i -= i & -i) {
        for (int j = y; j; j -= j & -j) {
            res += tr[i][j];
        }
    }
    return res;
}

LL sum(int x, int y) { // 前缀和查询：A[1..x][1..y]
    return (x + 1) * (y + 1) * sum(t1, x, y) //
           - (y + 1) * sum(t2, x, y)         //
           - (x + 1) * sum(t3, x, y)         //
           + sum(t4, x, y);
}

void update(int a, int b, int c, int d, LL delta) { // 矩形修改：A[a..c][b..d] += delta
    update(t1, a, b, delta), update(t1, a, d + 1, -delta);
    update(t1, c + 1, b, -delta), update(t1, c + 1, d + 1, delta);

    update(t2, a, b, a * delta), update(t2, a, d + 1, -a * delta);
    update(t2, c + 1, b, -(c + 1) * delta), update(t2, c + 1, d + 1, (c + 1) * delta);

    update(t3, a, b, b * delta), update(t3, a, d + 1, -(d + 1) * delta);
    update(t3, c + 1, b, -b * delta), update(t3, c + 1, d + 1, (d + 1) * delta);

    update(t4, a, b, a * b * delta), update(t4, a, d + 1, -a * (d + 1) * delta);
    update(t4, c + 1, b, -(c + 1) * b * delta), update(t4, c + 1, d + 1, (c + 1) * (d + 1) * delta);
}

LL sum(int a, int b, int c, int d) { // 矩形查询：A[a..c][b..d]
    return sum(c, d) - sum(a - 1, d) - sum(c, b - 1) + sum(a - 1, b - 1);
}
```

### 单点修改（赋值），区间最值查询

`tr[i] = max{A[i - lowbit(i) + 1 .. i]}`。max 不可逆（没有"差分"），查询不能像前缀和那样 `sum(r) - sum(l-1)`，只能从右往左扫描；降值修改也无法 O(1) 传播，必须重算覆盖节点。

```cpp
const int N = int(2e5) + 10;

int n;
int A[N];  // 原数组
int tr[N]; // tr[i] = max{A[i - lowbit(i) + 1 .. i]}

void update(int x, int v) { // 单点修改：A[x] = v
    bool up = v >= A[x];    // 必须和旧值 A[x] 比较，不能和 tr[x] 比
    A[x] = v;
    for (int i = x; i <= n; i += i & -i) {
        if (up) {
            tr[i] = std::max(tr[i], v); // 升值：取 max 即可
        } else {
            // 降值：重算整个管辖区间 = 叶子 A[i] + 子块 tr[i-1], tr[i-2], tr[i-4], ...
            tr[i] = A[i];
            for (int j = 1; j < (i & -i); j <<= 1) {
                tr[i] = std::max(tr[i], tr[i - j]);
            }
        }
    }
}

int query(int l, int r) { // 区间查询：max{A[l..r]}，从右往左扫描
    int res = 0;          // 值可非正时用 -INF
    for (int i = r; i >= l; i--) {
        int j = i - (i & -i) + 1; // tr[i] 管辖区间的左端点
        if (j >= l) {             // 整块在 [l, r] 内：用 tr[i] 并跳过整块
            res = std::max(res, tr[i]);
            i = j;
        } else { // 块越出左界：逐点取 A[i]
            res = std::max(res, A[i]);
        }
    }
    return res;
}
```

- 升/降值判断必须和旧值 `A[x]` 比较：若写成 `v >= tr[x]`，区间内其他位置比 `x` 大时会把降值误判为升值，导致 `tr` 不刷新。
- 建树逐个 `update(i, A[i])`，O(n log n)；多测记得 `std::fill(tr + 1, tr + 1 + n, 0)`。
- 降值修改 O(log² n)，其余 O(log n)；要稳定 O(log n) 请用线段树。
- 例题：HDU 1754 I Hate It。

## 线段树

### 区间修改，区间求和

```cpp
const int N = int(1e5) + 5;

using LL = long long;

int n;
LL A[N];

struct TreeNode {
    LL v, d; // v: 区间和, d: 懒标记（未下传的区间增量）
} tr[N * 4];

void pushup(int u) {
    tr[u].v = tr[u * 2].v + tr[u * 2 + 1].v;
}

void build(int u, int L, int R) { // 调用：build(1, 1, n)
    if (L == R) {
        tr[u] = {A[L], 0};
        return;
    }

    int mid = (L + R) / 2;
    build(u * 2, L, mid), build(u * 2 + 1, mid + 1, R);
    pushup(u);
}

void pushdown(int u, int L, int R) {
    if (tr[u].d) {
        int mid = (L + R) / 2;
        tr[u * 2].v += (mid - L + 1) * tr[u].d, tr[u * 2].d += tr[u].d;
        tr[u * 2 + 1].v += (R - mid) * tr[u].d, tr[u * 2 + 1].d += tr[u].d;
        tr[u].d = 0;
    }
}

void update(int u, int L, int R, int l, int r, LL d) { // 区间修改：A[l..r] += d
    if (L >= l && R <= r) {
        tr[u].v += (R - L + 1) * d, tr[u].d += d;
        return;
    }

    pushdown(u, L, R);
    int mid = (L + R) / 2;
    if (l <= mid) {
        update(u * 2, L, mid, l, r, d);
    }
    if (r > mid) {
        update(u * 2 + 1, mid + 1, R, l, r, d);
    }
    pushup(u);
}

LL query(int u, int L, int R, int l, int r) { // 区间查询：sum{A[l..r]}
    if (L >= l && R <= r) {
        return tr[u].v;
    }

    pushdown(u, L, R);
    int mid = (L + R) / 2;
    LL res = 0;
    if (l <= mid) {
        res += query(u * 2, L, mid, l, r);
    }
    if (r > mid) {
        res += query(u * 2 + 1, mid + 1, R, l, r);
    }
    return res;
}
```

- 递归前必须判断交集（`l <= mid` / `r > mid`），无交集的子树永远不命中返回条件，会递归不终止爆栈。
- 更新 `v` 时必须乘区间长度；`pushdown` 也因此要传入 `L, R` 计算子区间长度。
- `pushdown` 三要素：儿子 `v` 按子区间长度加值、儿子 `d` 累加、清空自己的 `d`。
- `update` 递归回溯后必须 `pushup`，否则父节点 `v` 过时。
- 例题：洛谷 P3372 线段树 1。

### Segment Tree Beats

#### 区间 chmin / 最大值 / 和

维护区间 `sum`、`max`、`second max`、`max count`，支持区间 `chmin`、区间最大值、区间和查询。

```cpp
const int N = int(1e6) + 5;

using LL = long long;

int n;
LL A[N];

struct TreeNode {
    LL sum, mx1, mx2; // sum: 区间和, mx1: 最大值, mx2: 次大值
    int cnt;          // mx1 出现次数
} tr[N * 4];

void pushup(int u) {
    int lc = u * 2, rc = u * 2 + 1;
    tr[u].sum = tr[lc].sum + tr[rc].sum;
    if (tr[lc].mx1 == tr[rc].mx1) {
        tr[u].mx1 = tr[lc].mx1;
        tr[u].cnt = tr[lc].cnt + tr[rc].cnt;
        tr[u].mx2 = std::max(tr[lc].mx2, tr[rc].mx2);
    } else {
        tr[u].mx1 = std::max(tr[lc].mx1, tr[rc].mx1);
        tr[u].cnt = tr[lc].mx1 > tr[rc].mx1 ? tr[lc].cnt : tr[rc].cnt;
        tr[u].mx2 = tr[lc].mx1 > tr[rc].mx1 ? std::max(tr[lc].mx2, tr[rc].mx1)
                                            : std::max(tr[lc].mx1, tr[rc].mx2);
    }
}

// 下传区间 chmin 标记：仅当子节点最大值更大时才更新
void pushdown(int u) {
    int lc = u * 2, rc = u * 2 + 1;
    if (tr[u].mx1 < tr[lc].mx1) {
        tr[lc].sum -= (tr[lc].mx1 - tr[u].mx1) * tr[lc].cnt;
        tr[lc].mx1 = tr[u].mx1;
    }
    if (tr[u].mx1 < tr[rc].mx1) {
        tr[rc].sum -= (tr[rc].mx1 - tr[u].mx1) * tr[rc].cnt;
        tr[rc].mx1 = tr[u].mx1;
    }
}

void build(int u, int L, int R) { // 调用：build(1, 1, n)
    if (L == R) {
        tr[u] = {A[L], A[L], -1, 1}; // 叶子次大值记为 -1
        return;
    }
    int mid = (L + R) / 2;
    build(u * 2, L, mid), build(u * 2 + 1, mid + 1, R);
    pushup(u);
}

// 区间 chmin：将 [l,r] 内元素与 a 取较小值
void chMin(int u, int L, int R, int l, int r, LL a) {
    if (a >= tr[u].mx1) {
        return; // a 不小于当前区间最大值，无需修改
    }
    if (L >= l && R <= r && a > tr[u].mx2) {
        // 直接拍扁当前区间的最大值
        tr[u].sum -= (tr[u].mx1 - a) * tr[u].cnt;
        tr[u].mx1 = a;
        return;
    }

    pushdown(u);
    int mid = (L + R) / 2;
    if (l <= mid) {
        chMin(u * 2, L, mid, l, r, a);
    }
    if (r > mid) {
        chMin(u * 2 + 1, mid + 1, R, l, r, a);
    }
    pushup(u);
}

LL queryMax(int u, int L, int R, int l, int r) { // 区间最大值
    if (L >= l && R <= r) {
        return tr[u].mx1;
    }
    pushdown(u);
    int mid = (L + R) / 2;
    LL res = -1;
    if (l <= mid) {
        res = std::max(res, queryMax(u * 2, L, mid, l, r));
    }
    if (r > mid) {
        res = std::max(res, queryMax(u * 2 + 1, mid + 1, R, l, r));
    }
    return res;
}

LL querySum(int u, int L, int R, int l, int r) { // 区间和
    if (L >= l && R <= r) {
        return tr[u].sum;
    }
    pushdown(u);
    int mid = (L + R) / 2;
    LL res = 0;
    if (l <= mid) {
        res += querySum(u * 2, L, mid, l, r);
    }
    if (r > mid) {
        res += querySum(u * 2 + 1, mid + 1, R, l, r);
    }
    return res;
}
```

- `pushup` 必须正确处理左右子节点最大值相等 / 不等的所有分支，并正确计算 `mx2` 与 `cnt`。
- `pushdown` 与查询前必须下传，否则子节点仍是旧值。
- 数值与 `sum` 都要用 `long long`，避免 `(mx1 - a) * cnt` 溢出。
- 例题：HDU 5306 Gorgeous Sequence。

#### 区间加 / chmin / 和 / 最大值 / 历史最大值

在上一版基础上额外支持**区间加**与**历史最大值**查询。核心是把“最大值集合”和“非最大值集合”分别打加法标记，并维护两类标记的历史最大，使得 chmin 下传时能正确还原历史信息。

```cpp
const int N = int(5e5) + 5;

using LL = long long;
const LL NEG_INF = -(1LL << 60);

int n;
int A[N];

struct TreeNode {
    LL sum;              // 区间和
    LL mx1, mx2, mxb;    // 当前最大值、次大值、历史最大值
    int cnt;             // 当前最大值出现次数
    LL da, db;           // 非最大值集合的加法标记，及其历史最大
    LL mxda, mxdb;       // 最大值集合的加法标记，及其历史最大
} tr[N * 4];

void pushup(int u) {
    int lc = u * 2, rc = u * 2 + 1;
    tr[u].sum = tr[lc].sum + tr[rc].sum;
    tr[u].mx1 = std::max(tr[lc].mx1, tr[rc].mx1);
    tr[u].mxb = std::max(tr[lc].mxb, tr[rc].mxb);
    if (tr[lc].mx1 > tr[rc].mx1) {
        tr[u].cnt = tr[lc].cnt;
        tr[u].mx2 = std::max(tr[lc].mx2, tr[rc].mx1);
    } else if (tr[lc].mx1 < tr[rc].mx1) {
        tr[u].cnt = tr[rc].cnt;
        tr[u].mx2 = std::max(tr[lc].mx1, tr[rc].mx2);
    } else {
        tr[u].cnt = tr[lc].cnt + tr[rc].cnt;
        tr[u].mx2 = std::max(tr[lc].mx2, tr[rc].mx2);
    }
}

// 对节点 u 覆盖的区间 [L,R] 打标记：
// 最大值集合加 mxda，非最大值集合加 da；同时维护两类标记的历史最大
void pushTag(int u, int L, int R, LL da, LL db, LL mxda, LL mxdb) {
    int len = R - L + 1;
    tr[u].sum += mxda * tr[u].cnt + da * (len - tr[u].cnt);
    tr[u].mxb = std::max(tr[u].mxb, tr[u].mx1 + mxdb);
    tr[u].mx1 += mxda;
    if (tr[u].mx2 != NEG_INF) {
        tr[u].mx2 += da;
    }
    tr[u].db = std::max(tr[u].db, tr[u].da + db);
    tr[u].da += da;
    tr[u].mxdb = std::max(tr[u].mxdb, tr[u].mxda + mxdb);
    tr[u].mxda += mxda;
}

void build(int u, int L, int R) {
    if (L == R) {
        tr[u] = {A[L], A[L], NEG_INF, A[L], 1, 0, 0, 0, 0};
        return;
    }
    int mid = (L + R) / 2;
    build(u * 2, L, mid), build(u * 2 + 1, mid + 1, R);
    tr[u].da = tr[u].db = tr[u].mxda = tr[u].mxdb = 0;
    pushup(u);
}

void pushdown(int u, int L, int R) {
    int mid = (L + R) / 2;
    int lc = u * 2, rc = u * 2 + 1;
    // 父节点去掉自身最大值集合标记后的底层最大值
    LL mx1u = tr[u].mx1 - tr[u].mxda;
    if (tr[lc].mx1 == mx1u) {
        pushTag(lc, L, mid, tr[u].da, tr[u].db, tr[u].mxda, tr[u].mxdb);
    } else {
        pushTag(lc, L, mid, tr[u].da, tr[u].db, tr[u].da, tr[u].db);
    }
    if (tr[rc].mx1 == mx1u) {
        pushTag(rc, mid + 1, R, tr[u].da, tr[u].db, tr[u].mxda, tr[u].mxdb);
    } else {
        pushTag(rc, mid + 1, R, tr[u].da, tr[u].db, tr[u].da, tr[u].db);
    }
    tr[u].da = tr[u].db = tr[u].mxda = tr[u].mxdb = 0;
}

// 区间加：A[l..r] += d
void add(int u, int L, int R, int l, int r, LL d) {
    if (L >= l && R <= r) {
        tr[u].sum += d * (R - L + 1);
        tr[u].mx1 += d;
        if (tr[u].mx2 != NEG_INF) {
            tr[u].mx2 += d;
        }
        tr[u].mxb = std::max(tr[u].mxb, tr[u].mx1);
        tr[u].da += d;
        tr[u].mxda += d;
        tr[u].db = std::max(tr[u].db, tr[u].da);
        tr[u].mxdb = std::max(tr[u].mxdb, tr[u].mxda);
        return;
    }
    pushdown(u, L, R);
    int mid = (L + R) / 2;
    if (l <= mid) {
        add(u * 2, L, mid, l, r, d);
    }
    if (r > mid) {
        add(u * 2 + 1, mid + 1, R, l, r, d);
    }
    pushup(u);
}

// 区间 chmin：A[l..r] = min(A[l..r], d)
void chMin(int u, int L, int R, int l, int r, LL d) {
    if (d >= tr[u].mx1) {
        return;
    }
    if (L >= l && R <= r && d > tr[u].mx2) {
        tr[u].sum -= (tr[u].mx1 - d) * tr[u].cnt;
        tr[u].mxda -= (tr[u].mx1 - d);
        tr[u].mx1 = d;
        return;
    }
    if (L == R) {
        tr[u].mx1 = std::min(tr[u].mx1, d);
        tr[u].sum = tr[u].mx1;
        return;
    }
    pushdown(u, L, R);
    int mid = (L + R) / 2;
    if (l <= mid) {
        chMin(u * 2, L, mid, l, r, d);
    }
    if (r > mid) {
        chMin(u * 2 + 1, mid + 1, R, l, r, d);
    }
    pushup(u);
}

LL querySum(int u, int L, int R, int l, int r) {
    if (L >= l && R <= r) {
        return tr[u].sum;
    }
    pushdown(u, L, R);
    int mid = (L + R) / 2;
    LL res = 0;
    if (l <= mid) {
        res += querySum(u * 2, L, mid, l, r);
    }
    if (r > mid) {
        res += querySum(u * 2 + 1, mid + 1, R, l, r);
    }
    return res;
}

LL queryMax(int u, int L, int R, int l, int r) {
    if (L >= l && R <= r) {
        return tr[u].mx1;
    }
    pushdown(u, L, R);
    int mid = (L + R) / 2;
    LL res = NEG_INF;
    if (l <= mid) {
        res = std::max(res, queryMax(u * 2, L, mid, l, r));
    }
    if (r > mid) {
        res = std::max(res, queryMax(u * 2 + 1, mid + 1, R, l, r));
    }
    return res;
}

LL queryHistMax(int u, int L, int R, int l, int r) {
    if (L >= l && R <= r) {
        return tr[u].mxb;
    }
    pushdown(u, L, R);
    int mid = (L + R) / 2;
    LL res = NEG_INF;
    if (l <= mid) {
        res = std::max(res, queryHistMax(u * 2, L, mid, l, r));
    }
    if (r > mid) {
        res = std::max(res, queryHistMax(u * 2 + 1, mid + 1, R, l, r));
    }
    return res;
}
```

- 需要同时维护“最大值集合”与“非最大值集合”两套加法标记，以及它们的历史最大值。
- `pushTag` 是核心：先按两类集合分别更新 `sum`、`mx1`、`mx2`，再用旧的当前标记加上新标记的历史最大来更新 `db`、`mxdb`。
- `pushdown` 时，子节点当前最大值若等于父节点“去掉自身最大值标记后的底层最大值”，则属于父节点的最大值集合，应接最大值集合标记；否则接非最大值集合标记。
- 区间 `chmin` 的拍扁条件与基础版相同：`d > mx2` 时只影响最大值集合。
- 例题：洛谷 P6242 线段树 3。
