# 数据结构模板

## 树状数组

### 单点修改，区间查询

```cpp
const int N = int(1e5) + 5;

using LL = long long;

int n;
LL tr[N];

void update(int x, LL d) { // 单点修改：A[x] += d
    for (int i = x; i <= n; i += i & -i) {
        tr[i] += d;
    }
}

LL sum(int x) { // 前缀和查询：A[1..x]
    LL res = 0;
    for (int i = x; i; i -= i & -i) {
        res += tr[i];
    }
    return res;
}

LL sum(int l, int r) { // 区间查询：A[l..r]
    return sum(r) - sum(l - 1);
}
```

- 用 `LL`（`long long`）存储和返回值，防止累加结果溢出（如逆序对计数最大约 $n^2/2$）。
- 多测时记得清空：`std::fill(tr, tr + n + 5, 0);`

### 区间修改，单点查询

设 `D` 为 `A` 的差分数组（`D[i] = A[i] - A[i - 1]`）。区间加 `[l, r]` 等价于 `D[l] += d`、`D[r + 1] -= d`，单点值即差分前缀和：`A[x] = D[1] + ... + D[x]`。

```cpp
const int N = int(1e5) + 5;

using LL = long long;

int n;
LL tr[N];

void update(int x, LL d) {
    for (int i = x; i <= n; i += i & -i) {
        tr[i] += d;
    }
}

LL sum(int x) {
    LL res = 0;
    for (int i = x; i; i -= i & -i) {
        res += tr[i];
    }
    return res;
}

void update(int l, int r, LL d) { // 区间修改：A[l..r] += d
    update(l, d), update(r + 1, -d);
}

// 单点查询：sum(x) 即 A[x] 的最终值
```

- 多测时记得清空：`std::fill(tr, tr + n + 5, 0);`

### 区间修改，区间查询

设 `D` 为 `A` 的差分数组，`PS(x) = A[1] + ... + A[x]` 为前缀和。展开可得 `PS(x) = Σ D[j]·(x - j + 1)`，用两个树状数组分别维护：

- `t1` 存 `D[j]` 的前缀和，记 `S1(x) = Σ D[j]`（`j ≤ x`）
- `t2` 存 `(j - 1) * D[j]` 的前缀和，记 `S2(x) = Σ (j - 1)·D[j]`（`j ≤ x`）

则前缀和公式：`PS(x) = x * S1(x) - S2(x)`，区间和 `[l, r] = PS(r) - PS(l - 1)`。

```cpp
const int N = int(1e5) + 5;

using LL = long long;

int n;
LL t1[N], t2[N];

void update(LL* tr, int x, LL d) {
    for (int i = x; i <= n; i += i & -i) {
        tr[i] += d;
    }
}

LL sum(LL* tr, int x) {
    LL res = 0;
    for (int i = x; i; i -= i & -i) {
        res += tr[i];
    }
    return res;
}

LL sum(int x) { // 前缀和查询：A[1..x]
    return x * sum(t1, x) - sum(t2, x);
}

void update(int l, int r, LL d) { // 区间修改：A[l..r] += d
    update(t1, l, d), update(t1, r + 1, -d);
    update(t2, l, (l - 1) * d), update(t2, r + 1, -r * d);
}

LL sum(int l, int r) { // 区间查询：A[l..r]
    return sum(r) - sum(l - 1);
}
```

- 建树时直接 `update(i, i, A[i])`，不要再手动差分。

### 二维：矩形修改，矩形查询

设 `D` 为矩阵 `A` 的二维差分数组（`A[i][j] = Σ_{p≤i, q≤j} D[p][q]`）。矩形加 `(a, b)-(c, d)` 等价于对 `D` 做 4 个单点更新：`D[a][b] += d`、`D[a][d+1] -= d`、`D[c+1][b] -= d`、`D[c+1][d+1] += d`。

二维前缀和展开：

$$
S(x, y) = \sum_{i \le x,\ j \le y} (x - i + 1)(y - j + 1) \cdot D_{i,j}
        = (x+1)(y+1) \sum D - (y+1) \sum i \cdot D - (x+1) \sum j \cdot D + \sum i \cdot j \cdot D
$$

用 4 个二维树状数组分别维护 `D`、`i·D`、`j·D`、`i·j·D` 的前缀和，矩形求和用前缀和容斥。

```cpp
const int N = 2060;

using LL = long long;

int n, m;
// t1 = D, t2 = i*D, t3 = j*D, t4 = i*j*D
LL t1[N][N], t2[N][N], t3[N][N], t4[N][N];

void update(LL tr[N][N], int x, int y, LL d) {
    for (int i = x; i <= n; i += i & -i) {
        for (int j = y; j <= m; j += j & -j) {
            tr[i][j] += d;
        }
    }
}

LL sum(LL tr[N][N], int x, int y) {
    LL res = 0;
    for (int i = x; i; i -= i & -i) {
        for (int j = y; j; j -= j & -j) {
            res += tr[i][j];
        }
    }
    return res;
}

LL sum(int x, int y) { // 前缀和查询：A[1..x][1..y]
    return (x + 1) * (y + 1) * sum(t1, x, y) //
           - (y + 1) * sum(t2, x, y)         //
           - (x + 1) * sum(t3, x, y)         //
           + sum(t4, x, y);
}

void update(int a, int b, int c, int d, LL delta) { // 矩形修改：A[a..c][b..d] += delta
    update(t1, a, b, delta), update(t1, a, d + 1, -delta);
    update(t1, c + 1, b, -delta), update(t1, c + 1, d + 1, delta);

    update(t2, a, b, a * delta), update(t2, a, d + 1, -a * delta);
    update(t2, c + 1, b, -(c + 1) * delta), update(t2, c + 1, d + 1, (c + 1) * delta);

    update(t3, a, b, b * delta), update(t3, a, d + 1, -(d + 1) * delta);
    update(t3, c + 1, b, -b * delta), update(t3, c + 1, d + 1, (d + 1) * delta);

    update(t4, a, b, a * b * delta), update(t4, a, d + 1, -a * (d + 1) * delta);
    update(t4, c + 1, b, -(c + 1) * b * delta), update(t4, c + 1, d + 1, (c + 1) * (d + 1) * delta);
}

LL sum(int a, int b, int c, int d) { // 矩形查询：A[a..c][b..d]
    return sum(c, d) - sum(a - 1, d) - sum(c, b - 1) + sum(a - 1, b - 1);
}
```

### 单点修改（赋值），区间最值查询

`tr[i] = max{A[i - lowbit(i) + 1 .. i]}`。max 不可逆（没有"差分"），查询不能像前缀和那样 `sum(r) - sum(l-1)`，只能从右往左扫描；降值修改也无法 O(1) 传播，必须重算覆盖节点。

```cpp
const int N = int(2e5) + 10;

int n;
int A[N];  // 原数组
int tr[N]; // tr[i] = max{A[i - lowbit(i) + 1 .. i]}

void update(int x, int v) { // 单点修改：A[x] = v
    bool up = v >= A[x];    // 必须和旧值 A[x] 比较，不能和 tr[x] 比
    A[x] = v;
    for (int i = x; i <= n; i += i & -i) {
        if (up) {
            tr[i] = std::max(tr[i], v); // 升值：取 max 即可
        } else {
            // 降值：重算整个管辖区间 = 叶子 A[i] + 子块 tr[i-1], tr[i-2], tr[i-4], ...
            tr[i] = A[i];
            for (int j = 1; j < (i & -i); j <<= 1) {
                tr[i] = std::max(tr[i], tr[i - j]);
            }
        }
    }
}

int query(int l, int r) { // 区间查询：max{A[l..r]}，从右往左扫描
    int res = 0;          // 值可非正时用 -INF
    for (int i = r; i >= l; i--) {
        int j = i - (i & -i) + 1; // tr[i] 管辖区间的左端点
        if (j >= l) {             // 整块在 [l, r] 内：用 tr[i] 并跳过整块
            res = std::max(res, tr[i]);
            i = j;
        } else { // 块越出左界：逐点取 A[i]
            res = std::max(res, A[i]);
        }
    }
    return res;
}
```

- 升/降值判断必须和旧值 `A[x]` 比较：若写成 `v >= tr[x]`，区间内其他位置比 `x` 大时会把降值误判为升值，导致 `tr` 不刷新。
- 建树逐个 `update(i, A[i])`，O(n log n)；多测记得 `std::fill(tr + 1, tr + 1 + n, 0)`。
- 降值修改 O(log² n)，其余 O(log n)；要稳定 O(log n) 请用线段树。
- 例题：HDU 1754 I Hate It。
